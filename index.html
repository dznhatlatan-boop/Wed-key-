"_" .. tostring(cycleIdx or 0) local hash = _SecretSalt for i = 1, #str do local b = string.byte(str, i) hash = (hash * 131 + b) % 89999999 end return tostring(10000000 + hash) end local function _IsValidKey(inputKey) local cleanKey = tostring(inputKey or ""):gsub("%D", "") if cleanKey == "" then return false end local baseCycle = _GetBaseCycleIndex() -- Cho phép khớp với chu kỳ hiện tại hoặc các chu kỳ gia hạn (tối đa 30 lần gia hạn) for offset = 0, 30 do local candidateKey = _TinhKey(_MePlr.UserId, baseCycle + offset) if cleanKey == candidateKey then return true end end return false end
